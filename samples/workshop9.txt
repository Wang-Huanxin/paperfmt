8. The following program is meant to create the image of a green rectangle.  However there are five 
 (5) errors in the program.  Identify each error and state what needs to be done to correct it. 
 
#include <iostream> 
#include <fstream> 
using namespace std; 
 
class pixel_class { 
private: 
  int red, green, blue; 
public: 
  void loaddata(int v1, int v2, int v3); 
  void datatofile(fstream ppmfile); 
  int getR() { return red; } 
  int getG() { return green; } 
  int getB() { return blue; } 
}; 
 
void createpicture(); 
 
pixel_class picture[500][300]; 
 
int main() { 
  int x, y; 
  fstream myfile; 
  createpicture(); 
  filename = "myImage.ppm"; 
  myfile.open(filename.c_str(), fstream::out); 
  myfile << "# " << filename << endl; 
  myfile << 250 << " " << 100 << endl; 
  myfile << 256 << endl; 
  for (y = 0; y < 300; y++) { 
    for (x = 0; x < 500; x++) { 
      picture[x][y].datatofile(myfile); 
    } 
    myfile << endl; 
  } 
  myfile.close(); 
} 
 
void createpicture() { 
  int x, y; 
  for (y = 0; y < 300; y++) { 
    for (x = 0; x < 500; x++) { 
      picture[x][y].loaddata(255, 153, 0); 
    } 
  } 
} 
 
//--------------- methods for the pixel_class ------------ 
void pixel_class::loaddata(int v1, int v2, int v3) { 
  red = v1; 
  green = v2; 
  blue = v3; 
} 
 
void pixel_class::datatofile(fstream & ppmfile) { 
  // write the data for one pixel to the ppm file 
  ppmfile << red << " " << green; 
  ppmfile << " " << blue << "  "; 
}
